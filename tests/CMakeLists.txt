# CTest runs executables; it is independent of the assertions/framework inside.
function(playground_add_test name)
    cmake_parse_arguments(PARSE_ARGV 1 test "" "SOURCE;MODULE;TIMEOUT" "LIBRARIES;LABELS")
    if(NOT test_SOURCE OR NOT test_MODULE)
        message(FATAL_ERROR "Tests require SOURCE and MODULE")
    endif()
    set(target playground_${name}_tests)
    add_executable(${target} ${test_SOURCE})
    playground_configure_cpp_target(${target})
    target_include_directories(${target} PRIVATE
        ${PROJECT_SOURCE_DIR}/include
        ${PROJECT_SOURCE_DIR}/tests)
    if(test_MODULE STREQUAL "ui")
        target_link_libraries(${target} PRIVATE playground_ui_core)
    elseif(test_MODULE STREQUAL "layout")
        target_link_libraries(${target} PRIVATE playground_layout)
    endif()
    if(test_LIBRARIES)
        target_link_libraries(${target} PRIVATE ${test_LIBRARIES})
    endif()
    if(NOT test_TIMEOUT)
        set(test_TIMEOUT 60)
    endif()
    add_test(NAME ${name} COMMAND ${target})
    set_tests_properties(${name} PROPERTIES
        LABELS "${test_MODULE};${test_LABELS}"
        TIMEOUT ${test_TIMEOUT})
    set_property(GLOBAL APPEND PROPERTY playground_test_targets ${target})
    set_property(GLOBAL APPEND PROPERTY playground_${test_MODULE}_test_targets ${target})
endfunction()
